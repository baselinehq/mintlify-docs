---
title: 'VM Recommendations'
openapi: 'GET /v1/vm_recommendations'
---

## Response

<ResponseField name="vm" type="string">
  The name of the virtual machine.
</ResponseField>

<ResponseField name="process" type="string">
  The process name or identifier.
</ResponseField>

<ResponseField name="processGroup" type="string">
  The group classification of the process.
  
  Possible values: `kubernetes`, `docker`, `system`, `user`, `unknown`
</ResponseField>

<ResponseField name="cgroup" type="string">
  The cgroup path for the process.
</ResponseField>

<ResponseField name="costDriver" type="string">
  The primary cost driver for this VM.
</ResponseField>

<ResponseField name="currentSpend" type="number">
  Current spending amount in dollars.
</ResponseField>

<ResponseField name="creationTime" type="string">
  The creation time of the recommendation.
  
  Example: `"2023-10-01T00:00:00Z"`
</ResponseField>

<ResponseField name="cpu" type="object">
  CPU resource recommendations including baseline, current, and recommended usage.
  
  <Expandable title="CPU Recommendations">
    <ResponseField name="baseline" type="object">
      Historical baseline usage (mean/p99) that recommendations are based on.
      
      <Expandable title="Baseline Details">
        <ResponseField name="value" type="number">
          CPU usage value.
        </ResponseField>
        
        <ResponseField name="costInDollars" type="number">
          Cost in dollars for the baseline usage.
        </ResponseField>
      </Expandable>
    </ResponseField>
    
    <ResponseField name="current" type="object">
      Real-time current usage since reconcile time.
      
      <Expandable title="Current Details">
        <ResponseField name="value" type="number">
          Current CPU usage value.
        </ResponseField>
        
        <ResponseField name="costInDollars" type="number">
          Cost in dollars for the current usage.
        </ResponseField>
      </Expandable>
    </ResponseField>
    
    <ResponseField name="recommended" type="object">
      Recommended CPU usage based on analysis.
      
      <Expandable title="Recommended Details">
        <ResponseField name="value" type="number">
          Recommended CPU usage value.
        </ResponseField>
        
        <ResponseField name="costInDollars" type="number">
          Projected cost in dollars for the recommended usage.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="memory" type="object">
  Memory resource recommendations including baseline, current, and recommended usage.
  
  <Expandable title="Memory Recommendations">
    <ResponseField name="baseline" type="object">
      Historical baseline memory usage.
    </ResponseField>
    
    <ResponseField name="current" type="object">
      Real-time current memory usage.
    </ResponseField>
    
    <ResponseField name="recommended" type="object">
      Recommended memory usage based on analysis.
    </ResponseField>
  </Expandable>
</ResponseField>
