---
title: 'VM Summaries'
openapi: 'GET /v1/vm_summaries'
---


## Response

<ResponseField name="vm" type="string">
  The name of the virtual machine.
</ResponseField>

<ResponseField name="creationTime" type="string">
  The creation time of the summary.
  
  Example: `"2023-10-01T00:00:00Z"`
</ResponseField>

<ResponseField name="totalCurrentCPU" type="number">
  Total current CPU usage across all processes on the VM.
</ResponseField>

<ResponseField name="totalCurrentRAM" type="number">
  Total current RAM usage across all processes on the VM.
</ResponseField>

<ResponseField name="totalRecommendedCPU" type="number">
  Total recommended CPU allocation based on usage analysis.
</ResponseField>

<ResponseField name="totalRecommendedRAM" type="number">
  Total recommended RAM allocation based on usage analysis.
</ResponseField>

<ResponseField name="expectedUtilisationCPU" type="number">
  Expected CPU utilization percentage based on historical data.
</ResponseField>

<ResponseField name="expectedUtilisationRAM" type="number">
  Expected RAM utilization percentage based on historical data.
</ResponseField>
