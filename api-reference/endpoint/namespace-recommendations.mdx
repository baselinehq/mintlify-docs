---
title: 'Namespace Recommendations'
openapi: 'GET /v1/namespace_recommendations'
---

## Query Parameters

<SchemaField name="cluster" type="string">
  Cluster name to filter recommendations by.
  
  Example: `production-cluster`
</SchemaField>

<SchemaField name="namespace" type="string">
  Namespace to filter recommendations by.
  
  Example: `default`
</SchemaField>

<SchemaField name="start_date" type="string" required>
  Start date in RFC3339 format.
  
  Example: `2023-10-01T00:00:00Z`
</SchemaField>

<SchemaField name="end_date" type="string" required>
  End date in RFC3339 format.
  
  Example: `2023-10-31T23:59:59Z`
</SchemaField>

## Response

<ResponseField name="APIVersion" type="string">
  API version of the recommendation.
</ResponseField>

<ResponseField name="Cluster" type="string">
  The name of the Kubernetes cluster.
</ResponseField>

<ResponseField name="Namespace" type="string">
  The name of the Kubernetes namespace.
</ResponseField>

<ResponseField name="CreationTime" type="string">
  The creation time of the recommendation.
  
  Example: `"2023-10-01T00:00:00Z"`
</ResponseField>

<ResponseField name="CurrentSpend" type="number">
  Current spending amount in dollars.
</ResponseField>

<ResponseField name="NewSpend" type="number">
  Projected spending amount after implementing recommendations.
</ResponseField>

<ResponseField name="TotalResourceSavingsAmount" type="number">
  Total amount of savings in dollars from implementing recommendations.
</ResponseField>

<ResponseField name="ExpectedUtilisationPercent" type="object">
  Expected utilization percentages for CPU and memory.
  
  <Expandable title="Utilization Percentages">
    <ResponseField name="cpu" type="number">
      Expected CPU utilization percentage.
    </ResponseField>
    
    <ResponseField name="memory" type="number">
      Expected memory utilization percentage.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="ContainerRecommendations" type="object">
  Detailed recommendations for containers within the namespace.
</ResponseField>

<ResponseField name="ResourceQuotaRecommendations" type="object">
  Recommendations for resource quotas including CPU and memory limits.
  
  <Expandable title="Resource Quota Recommendations">
    <ResponseField name="CPU" type="object">
      CPU quota recommendations with current and recommended values.
    </ResponseField>
    
    <ResponseField name="Memory" type="object">
      Memory quota recommendations with current and recommended values.
    </ResponseField>
  </Expandable>
</ResponseField>
